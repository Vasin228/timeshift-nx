cmake_minimum_required(VERSION 3.20)
project(timeshift_renut_adapter LANGUAGES CXX)
# This target is an offline CPU adapter fixture, not a game or renderer build.
add_library(timeshift_device_bootstrap STATIC adapter/device_bootstrap.cpp)
target_include_directories(timeshift_device_bootstrap PUBLIC adapter)
target_compile_features(timeshift_device_bootstrap PUBLIC cxx_std_20)
option(TIMESHIFT_BUILD_PRIVATE_FIXTURE "Build against four locally extracted guest functions" OFF)
if(TIMESHIFT_BUILD_PRIVATE_FIXTURE)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/analysis/fixture/guest_subset.inc")
    message(FATAL_ERROR "Run tools/prepare_bootstrap_fixture.py locally first")
  endif()
  add_executable(device_bootstrap_fixture tests/device_bootstrap_fixture.cpp)
  target_include_directories(device_bootstrap_fixture PRIVATE analysis/fixture)
  target_link_libraries(device_bootstrap_fixture PRIVATE timeshift_device_bootstrap)
  enable_testing()
  add_test(NAME device_bootstrap_fixture COMMAND device_bootstrap_fixture)
  set_tests_properties(device_bootstrap_fixture PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()

option(TIMESHIFT_BUILD_SHADER_TRANSLATOR "Build the pinned MIT XenosRecomp host tool" OFF)
if(TIMESHIFT_BUILD_SHADER_TRANSLATOR)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This isolated translator setup currently targets the WSL Linux host")
  endif()
  set(TIMESHIFT_TRANSLATOR_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/upstream/sa2-nx/app/thirdparty/reblue-XenosRecomp")
  set(TIMESHIFT_DXC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/build/deps/dxc-1.8.2505.1" CACHE PATH "Verified official DXC host package")
  if(NOT EXISTS "${TIMESHIFT_DXC_ROOT}/lib/libdxcompiler.so")
    message(FATAL_ERROR "Run tools/prepare_dxc.py first")
  endif()
  add_library(Microsoft::DirectXShaderCompiler SHARED IMPORTED GLOBAL)
  set_target_properties(Microsoft::DirectXShaderCompiler PROPERTIES
    IMPORTED_LOCATION "${TIMESHIFT_DXC_ROOT}/lib/libdxcompiler.so"
    INTERFACE_INCLUDE_DIRECTORIES "${TIMESHIFT_TRANSLATOR_SOURCE}/thirdparty/dxc-bin/inc")
  set(FMT_DOC OFF CACHE BOOL "" FORCE)
  set(FMT_TEST OFF CACHE BOOL "" FORCE)
  set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(XENOS_RECOMP_RENUT ON CACHE BOOL "" FORCE)
  add_subdirectory("${TIMESHIFT_TRANSLATOR_SOURCE}" "${CMAKE_BINARY_DIR}/xenos-recomp")
  target_compile_features(XenosRecomp PRIVATE cxx_std_20)
  target_compile_definitions(XenosRecomp PRIVATE TIMESHIFT_RECOMP)
  # Keep format/instruction assertions active while porting an untested title.
  target_compile_options(XenosRecomp PRIVATE -UNDEBUG)
endif()
